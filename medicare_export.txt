===== FILE: src/controllers/accountController.js =====
const patientService = require('../services/patientService');
const User = require('../models/User');
const { startSession } = require('./authController');
const { pick, toErrors, isFormError } = require('./patientController');
const { fieldErrors } = require('../validators/common');
const { toInputDate } = require('../utils/dates');

function renderRegister(res, status, values, errors = {}) {
  res.status(status).render('pages/account/register', { title: 'Créer un compte', values, errors });
}

function showRegister(req, res) {
  renderRegister(res, 200, pick({}));
}

async function register(req, res, next) {
  const values = pick(req.body);
  const errors = fieldErrors(req);
  if (Object.keys(errors).length) return renderRegister(res, 422, values, errors);
  try {
    const patient = await patientService.create({ ...values, password: req.body.password });
    const user = await User.findById(patient.user);
    return startSession(req, res, next, user, `Bienvenue, ${user.firstName}. Votre compte a été créé.`);
  } catch (err) {
    if (isFormError(err)) return renderRegister(res, err.status, values, toErrors(err));
    return next(err);
  }
}

function renderProfile(res, status, values, errors = {}) {
  res.status(status).render('pages/account/profile', { title: 'Mon profil', values, errors });
}

async function showProfile(req, res) {
  const patient = await patientService.getByUser(req.session.user.id);
  renderProfile(res, 200, { ...pick(patient.user), birthDate: toInputDate(patient.birthDate), phone: patient.phone });
}

async function updateProfile(req, res) {
  const patient = await patientService.getByUser(req.session.user.id);
  const values = pick(req.body);
  const errors = fieldErrors(req);
  if (Object.keys(errors).length) return renderProfile(res, 422, values, errors);
  try {
    const { user } = await patientService.update(patient._id, { ...values, password: req.body.password });
    req.session.user.fullName = user.fullName; // keep the header in sync
    req.flash('success', 'Vos informations ont été enregistrées.');
    return res.redirect('/profile');
  } catch (err) {
    if (isFormError(err)) return renderProfile(res, err.status, values, toErrors(err));
    throw err;
  }
}

module.exports = { showRegister, register, showProfile, updateProfile };

===== FILE: src/controllers/patientController.js =====
const patientService = require('../services/patientService');
const { fieldErrors } = require('../validators/common');
const { toInputDate } = require('../utils/dates');

const FIELDS = ['firstName', 'lastName', 'email', 'birthDate', 'phone'];
const pick = (body) => Object.fromEntries(FIELDS.map((f) => [f, (body[f] || '').toString().trim()]));
const toErrors = (err) => (err.field ? { [err.field]: err.message } : { form: err.message });
const isFormError = (err) => err.status === 409 || err.status === 422;
const fullName = (patient) => `${patient.user.firstName} ${patient.user.lastName}`;

function renderForm(res, status, { patient = null, values, errors = {} }) {
  res.status(status).render('pages/patients/form', {
    title: patient ? 'Modifier le patient' : 'Nouveau patient',
    patient,
    values,
    errors,
  });
}

async function index(req, res) {
  res.render('pages/patients/index', { title: 'Patients', patients: await patientService.list() });
}

async function show(req, res) {
  const patient = await patientService.get(req.params.id);
  res.render('pages/patients/show', { title: fullName(patient), patient });
}

function newForm(req, res) {
  renderForm(res, 200, { values: pick({}) });
}

async function create(req, res) {
  const values = pick(req.body);
  const errors = fieldErrors(req);
  if (Object.keys(errors).length) return renderForm(res, 422, { values, errors });
  try {
    const patient = await patientService.create({ ...values, password: req.body.password });
    req.flash('success', 'Patient créé.');
    return res.redirect(`/patients/${patient.id}`);
  } catch (err) {
    if (isFormError(err)) return renderForm(res, err.status, { values, errors: toErrors(err) });
    throw err;
  }
}

async function editForm(req, res) {
  const patient = await patientService.get(req.params.id);
  const values = { ...pick(patient.user), birthDate: toInputDate(patient.birthDate), phone: patient.phone };
  renderForm(res, 200, { patient, values });
}

async function update(req, res) {
  const patient = await patientService.get(req.params.id);
  const values = pick(req.body);
  const errors = fieldErrors(req);
  if (Object.keys(errors).length) return renderForm(res, 422, { patient, values, errors });
  try {
    await patientService.update(req.params.id, { ...values, password: req.body.password });
    req.flash('success', 'Patient modifié.');
    return res.redirect(`/patients/${req.params.id}`);
  } catch (err) {
    if (isFormError(err)) return renderForm(res, err.status, { patient, values, errors: toErrors(err) });
    throw err;
  }
}

async function remove(req, res) {
  try {
    const { patient, deactivated } = await patientService.remove(req.params.id);
    req.flash('success', deactivated
      ? `Patient « ${fullName(patient)} » désactivé : son historique de rendez-vous est conservé.`
      : `Patient « ${fullName(patient)} » supprimé.`);
  } catch (err) {
    if (err.status !== 409) throw err;
    req.flash('danger', err.message);
  }
  res.redirect('/patients');
}

async function reactivate(req, res) {
  const patient = await patientService.reactivate(req.params.id);
  req.flash('success', `Patient « ${fullName(patient)} » réactivé.`);
  res.redirect(`/patients/${req.params.id}`);
}

// Copies local fonts and builds the icon sprite. Output is committed,
// so this only needs to run when the icon list or fonts change.
const fs = require('fs');
const path = require('path');

const root = path.join(__dirname, '..');
const assets = path.join(root, 'src/public/assets');

const ICONS = [
  'menu', 'heart-pulse', 'layout-dashboard', 'calendar-days', 'stethoscope',
  'users', 'user-round', 'tags', 'settings', 'clock', 'plus', 'log-out',
  'circle-user', 'calendar-plus', 'shield-check', 'arrow-up-right', 'calendar-heart',
  'lock-keyhole', 'heart-handshake', 'clock-3', 'chevron-left', 'chevron-right',
  'x', 'pencil', 'trash-2', 'circle-check', 'circle-alert', 'info', 'calendar-check',
];
const FONT_WEIGHTS = [400, 500, 600];

function buildSprite() {
  const symbols = ICONS.map((name) => {
    const svg = fs.readFileSync(path.join(root, 'node_modules/lucide-static/icons', `${name}.svg`), 'utf8');
    const inner = svg.replace(/<svg[^>]*>/, '').replace('</svg>', '').replace(/<!--[\s\S]*?-->/g, '').trim();
    return `<symbol id="${name}" viewBox="0 0 24 24">${inner}</symbol>`;
  });
  const sprite = `<svg xmlns="http://www.w3.org/2000/svg">\n${symbols.join('\n')}\n</svg>\n`;
  fs.writeFileSync(path.join(assets, 'icons.svg'), sprite);
}

function copyFonts() {
  const src = path.join(root, 'node_modules/@fontsource/ibm-plex-sans/files');
  const dest = path.join(assets, 'fonts');
  for (const w of FONT_WEIGHTS) {
    const file = `ibm-plex-sans-latin-${w}-normal.woff2`;
    fs.copyFileSync(path.join(src, file), path.join(dest, file));
  }
}

buildSprite();
copyFonts();
console.log(`Sprite: ${ICONS.length} icons, fonts: ${FONT_WEIGHTS.join(', ')}`);

  ],
  patient: [
    { href: '/', label: 'Accueil', icon: 'layout-dashboard' },
    { href: '/appointments/new', label: 'Prendre rendez-vous', icon: 'plus' },
    { href: '/appointments', label: 'Mes rendez-vous', icon: 'calendar-days' },
    { href: '/profile', label: 'Mon profil', icon: 'user-round' },
  ],

function matches(href, path) {
  return href === '/' ? path === '/' : path === href || path.startsWith(`${href}/`);
}

// Returns the role's menu with `active` set on the single best (longest) match.
function getNavigation(role, path) {
  const items = NAVIGATION[role] || [];
  const best = items
    .filter((item) => matches(item.href, path))
    .sort((a, b) => b.href.length - a.href.length)[0];
  return items.map((item) => ({ ...item, active: item === best }));
}

module.exports = { ROLE_LABELS, getNavigation };

// Variables available in every Pug template.
function viewLocals(req, res, next) {
  const user = (req.session && req.session.user) || null;

  res.locals.currentUser = user;
  res.locals.roleLabel = user ? ROLE_LABELS[user.role] : null;
  res.locals.nav = user ? getNavigation(user.role, req.path) : [];
  res.locals.formatDate = formatDate;
  res.locals.formatDay = formatDay;
  res.locals.formatTime = formatTime;
  res.locals.appVersion = pkg.version;
  res.locals.currentYear = new Date().getFullYear();
  next();
}
