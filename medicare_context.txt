// Sidebar behaviour. Loaded in <head> so the initial state is set before first paint.
(function () {
  var root = document.documentElement;
  var KEY = 'medicare.sidebar';
  var desktop = window.matchMedia('(min-width: 62rem)');

  function read() { try { return localStorage.getItem(KEY); } catch (e) { return null; } }
  function write(value) { try { localStorage.setItem(KEY, value); } catch (e) { /* storage unavailable */ } }

  // Desktop remembers the choice (open by default); mobile always starts closed.
  function initialState() { return desktop.matches && read() !== 'closed' ? 'open' : 'closed'; }

  root.dataset.sidebar = initialState();

  document.addEventListener('DOMContentLoaded', function () {
    var toggle = document.getElementById('sidebar-toggle');
    var backdrop = document.getElementById('sidebar-backdrop');
    if (!toggle) return;

    function set(state) {
      root.dataset.sidebar = state;
      toggle.setAttribute('aria-expanded', String(state === 'open'));
      if (desktop.matches) write(state);
    }

    toggle.setAttribute('aria-expanded', String(root.dataset.sidebar === 'open'));
    toggle.addEventListener('click', function () {
      set(root.dataset.sidebar === 'open' ? 'closed' : 'open');
    });
    if (backdrop) backdrop.addEventListener('click', function () { set('closed'); });
    document.addEventListener('keydown', function (event) {
      if (event.key === 'Escape' && !desktop.matches && root.dataset.sidebar === 'open') {
        set('closed');
        toggle.focus();
      }
    });
    desktop.addEventListener('change', function () {
      root.dataset.sidebar = initialState();
      toggle.setAttribute('aria-expanded', String(root.dataset.sidebar === 'open'));
    });
  });
})();

// Accessible, dependency-free carousel for the home page.
(function () {
  document.addEventListener('DOMContentLoaded', function () {
    var carousel = document.querySelector('[data-carousel]');
    if (!carousel) return;
    var slides = Array.from(carousel.querySelectorAll('.feature-slide'));
    var dots = Array.from(document.querySelectorAll('[data-carousel-dot]'));
    var current = 0;
    function show(index) {
      current = (index + slides.length) % slides.length;
      slides.forEach(function (slide, i) {
        var active = i === current;
        slide.classList.toggle('is-active', active);
        slide.setAttribute('aria-hidden', String(!active));
      });
      dots.forEach(function (dot, i) {
        var active = i === current;
        dot.classList.toggle('is-active', active);
        if (active) dot.setAttribute('aria-current', 'true');
        else dot.removeAttribute('aria-current');
      });
    }
    var previous = document.querySelector('[data-carousel-prev]');
    var next = document.querySelector('[data-carousel-next]');
    if (previous) previous.addEventListener('click', function () { show(current - 1); });
    if (next) next.addEventListener('click', function () { show(current + 1); });
    dots.forEach(function (dot, i) { dot.addEventListener('click', function () { show(i); }); });
    show(0);
  });
})();

// Notifications (dismiss, auto-dismiss) and confirmation dialog for forms with data-confirm.
// Must be loaded before forms.js so a pending confirmation cancels the loading state.
(function () {
  document.addEventListener('click', function (event) {
    var close = event.target.closest('[data-dismiss]');
    if (!close) return;
    var alert = close.closest('.alert');
    if (alert) alert.remove();
  });

  document.querySelectorAll('.alert[data-autodismiss]').forEach(function (alert) {
    setTimeout(function () { alert.remove(); }, 6000);
  });

  var dialog = document.getElementById('confirm-dialog');
  if (!dialog || typeof dialog.showModal !== 'function') return;
  var pending = null;

  document.addEventListener('submit', function (event) {
    var form = event.target;
    if (!form.dataset.confirm || form.dataset.confirmed) return;
    event.preventDefault();
    pending = form;
    document.getElementById('confirm-title').textContent = form.dataset.confirmTitle || 'Confirmer';
    document.getElementById('confirm-message').textContent = form.dataset.confirm;
    document.getElementById('confirm-accept').textContent = form.dataset.confirmLabel || 'Confirmer';
    dialog.returnValue = '';
    dialog.showModal();
  });

  dialog.addEventListener('close', function () {
    var form = pending;
    pending = null;
    if (dialog.returnValue !== 'confirm' || !form) return;
    form.dataset.confirmed = 'true';
    if (form.requestSubmit) form.requestSubmit();
    else form.submit();
  });
})();

const User = require('../models/User');
const HttpError = require('../utils/httpError');
const { COOKIE_NAME } = require('../config/session');

const isApi = (req) => req.originalUrl.startsWith('/api/');

function redirectIfAuthenticated(req, res, next) {
  if (req.session && req.session.user) return res.redirect('/');
  return next();
}

// Requires a logged-in, still-active user. The role is re-read from the database
// on every request, so deactivating a user or changing a role applies immediately.
async function requireAuth(req, res, next) {
  const sessionUser = req.session && req.session.user;
  if (!sessionUser) {
    return isApi(req) ? next(new HttpError(401, 'Authentification requise.')) : res.redirect('/login');
  }
  try {
    const user = await User.findById(sessionUser.id).select('active role');
    if (!user || !user.active) {
      return req.session.destroy(() => {
        res.clearCookie(COOKIE_NAME);
        if (isApi(req)) return next(new HttpError(401, 'Authentification requise.'));
        return res.redirect('/login');
      });
    }
    sessionUser.role = user.role;
    return next();
  } catch (err) {
    return next(err);
  }
}

// Use after requireAuth: requireRole('admin') or requireRole('admin', 'doctor').
function requireRole(...roles) {
  return (req, res, next) => {
    const user = req.session && req.session.user;
    if (!user || !roles.includes(user.role)) {
      return next(new HttpError(403, "Vous n'avez pas accès à cette page."));
    }
    return next();
  };
}

module.exports = { redirectIfAuthenticated, requireAuth, requireRole };

const config = require('../config/env');

function notFound(req, res, next) {
  const err = new Error('Page introuvable');
  err.status = 404;
  next(err);
}

function errorHandler(err, req, res, next) {
  const status = err.status || 500;
  const message = status === 500 && config.nodeEnv === 'production'
    ? 'Une erreur interne est survenue.'
    : err.message;

  if (status === 500) console.error(err);

  if (req.originalUrl.startsWith('/api/')) {
    const body = { status, message };
    if (err.type === 'entity.parse.failed') body.message = 'Corps de requête JSON invalide.';
    const fields = err.fields || (err.field ? { [err.field]: err.message } : null);
    if (fields) body.fields = fields;
    return res.status(status).json({ error: body });
  }
  res.status(status).render('errors/error', { title: `Erreur ${status}`, status, message });
}

module.exports = { notFound, errorHandler };

// One-time messages that survive a redirect: req.flash('success', 'Message').
function flash(req, res, next) {
  req.flash = (type, message) => {
    req.session.flash = [...(req.session.flash || []), { type, message }];
  };
  res.locals.flash = req.session.flash || [];
  delete req.session.flash;
  next();
}

module.exports = flash;

const bcrypt = require('bcryptjs');
const User = require('../models/User');
const HttpError = require('../utils/httpError');

const SALT_ROUNDS = 10;
const MIN_PASSWORD_LENGTH = 8;

function checkPassword(password) {
  if (typeof password !== 'string' || password.length < MIN_PASSWORD_LENGTH) {
    throw new HttpError(422, `Le mot de passe doit contenir au moins ${MIN_PASSWORD_LENGTH} caractères.`, 'password');
  }
}

function translate(err) {
  if (err.code === 11000) return new HttpError(409, 'Cette adresse e-mail est déjà utilisée.', 'email');
  if (err.name === 'ValidationError') return new HttpError(422, "Les informations de l'utilisateur sont invalides.");
  return err;
}

async function createUser({ email, password, role, firstName, lastName }) {
  checkPassword(password);
  const passwordHash = await bcrypt.hash(password, SALT_ROUNDS);
  try {
    return await User.create({ email, passwordHash, role, firstName, lastName });
  } catch (err) {
    throw translate(err);
  }
}

// `password` is optional: leave it empty to keep the current one.
async function updateUser(id, { email, password, firstName, lastName }) {
  const user = await User.findById(id).select('+passwordHash');
  if (!user) throw new HttpError(404, 'Utilisateur introuvable.');
  if (password) {
    checkPassword(password);
    user.passwordHash = await bcrypt.hash(password, SALT_ROUNDS);
  }
  Object.assign(user, { email, firstName, lastName });
  try {
    return await user.save();
  } catch (err) {
    throw translate(err);
  }
}

async function setActive(id, active) {
  const result = await User.updateOne({ _id: id }, { active });
  if (!result.matchedCount) throw new HttpError(404, 'Utilisateur introuvable.');
}

module.exports = { createUser, updateUser, setActive, MIN_PASSWORD_LENGTH };

  return date;
}

async function list() {
  const patients = await populate(Patient.find()).lean();
  const key = (p) => `${p.user.lastName} ${p.user.firstName}`;
  return patients.sort((a, b) => collator.compare(key(a), key(b)));
}

async function get(id) {
  if (!mongoose.isValidObjectId(id)) throw new HttpError(404, NOT_FOUND);
  const patient = await populate(Patient.findById(id)).lean();
  if (!patient) throw new HttpError(404, NOT_FOUND);
  return patient;
}

async function getByUser(userId) {
  const patient = await populate(Patient.findOne({ user: userId })).lean();
  if (!patient) throw new HttpError(404, NOT_FOUND);
  return patient;
}

async function create({ firstName, lastName, email, password, birthDate, phone }) {
  const date = checkBirthDate(birthDate);
  const user = await userService.createUser({ email, password, firstName, lastName, role: 'patient' });
  try {
    return await Patient.create({ user: user.id, birthDate: date, phone });
  } catch (err) {
    await User.deleteOne({ _id: user.id }); // no orphan account if the profile cannot be saved
    if (err.name === 'ValidationError') throw new HttpError(422, 'Les informations du patient sont invalides.');
    throw err;
  }
}

async function update(id, { firstName, lastName, email, password, birthDate, phone }) {
  if (!mongoose.isValidObjectId(id)) throw new HttpError(404, NOT_FOUND);
  const patient = await Patient.findById(id);
  if (!patient) throw new HttpError(404, NOT_FOUND);
  const date = checkBirthDate(birthDate);
  const user = await userService.updateUser(patient.user, { email, password, firstName, lastName });
  Object.assign(patient, { birthDate: date, phone });
  await patient.save();
  return { patient, user };
}

// Deletes a patient with no appointment history; deactivates one whose history must be kept;
// refuses while upcoming appointments exist.
async function remove(id) {
  const patient = await get(id);
  const plan = await removalPlan({ patient: patient._id });
  if (plan.action === 'refuse') {
    throw new HttpError(409, `Ce patient a ${plan.upcoming} rendez-vous à venir. Annulez-les avant de le supprimer.`);
  }
  if (plan.action === 'deactivate') {
    await User.updateOne({ _id: patient.user._id }, { active: false });
    return { patient, deactivated: true };
  }
  await Patient.deleteOne({ _id: id });
  await User.deleteOne({ _id: patient.user._id });
  return { patient, deactivated: false };
}

async function reactivate(id) {
  const patient = await get(id);
  await userService.setActive(patient.user._id, true);
  return patient;
}

module.exports = { list, get, getByUser, create, update, remove, reactivate };

// Starts a logged-in session. A new session id on login prevents session fixation.
function establishSession(req, user) {
  return new Promise((resolve, reject) => {
    req.session.regenerate((regenError) => {
      if (regenError) return reject(regenError);
      req.session.user = { id: user.id, fullName: user.fullName, role: user.role };
      return req.session.save((saveError) => (saveError ? reject(saveError) : resolve()));
    });
  });
}

module.exports = { establishSession };

- Specialties (admin only): list, create, edit, delete; unique ignoring case/accents (collation fr strength 1)
- Doctors (admin only): list, view, create, edit, delete; creating a doctor creates a `doctor` login account (rolled back if the profile fails); empty password on edit keeps the old one; licence number unique
- A specialty used by a doctor cannot be deleted
- Patients: admin CRUD; public self-registration at `/register` (always role `patient`, logs the user in); patients edit their own info at `/profile` (header name kept in sync); birth date validated (no future, >= 1900) and shown dd/mm/yyyy
- Availability (Phase 5): `AvailabilitySlot` (doctor, start, end, status free/booked; unique {doctor,start}). Doctors manage own slots, admins any doctor's via `?doctor=<id>` (also linked from the doctor page). One form adds a range split into 15/20/30/45/60-min slots or one single slot (max 48); overlaps refused (adjacent OK); edit/delete only while free; times typed in clinic time (`CLINIC_TIMEZONE`) and stored in UTC; list shows upcoming slots grouped by day
- Appointments step 1: `Appointment` (slot, doctor, patient, start/end copied from slot, status confirmed|cancelled, reason<=300; partial unique index on slot where confirmed). Patient flow: `/appointments/new` (pick specialty -> doctors with free-slot counts) -> `/appointments/new/:doctorId` (free upcoming slots by day) -> `/appointments/book/:slotId` (confirm + optional reason) -> POST `/appointments`. Booking = atomic `findOneAndUpdate({status:'free', start>now}) -> booked`, then create appointment, slot released if creation fails. A patient cannot hold two overlapping appointments. `/appointments` lists upcoming + history, scoped by role (patient own, doctor own, admin all)
- Appointments step 2: POST `/appointments/:id/cancel` (patient own, doctor own, admin any; others get 404). Appointment is cancelled atomically first (`status:'confirmed', start>now`), then its slot is released (`status:'booked'` guard), so a double cancel cannot free a re-booked slot. Past or already-cancelled appointments are refused with a message. `cancelledAt`/`cancelledBy` kept; history shows who cancelled
- Appointments step 3: `/appointments/:id/edit` + POST `/appointments/:id` (patient own, admin any; doctors 403, others 404). Move to another free slot of the SAME doctor and/or change the reason. Safe order: claim new slot -> switch appointment (guarded by `status:'confirmed'` and its current slot) -> release old slot; any failure releases the claimed slot. A slot that vanished or was taken gives a normal message, not a 404 page. Tests assert the invariant (booked slots == confirmed appointments' slots, times match) under concurrent move/move and move/cancel
- Appointments step 4a: `/appointments/day?date=YYYY-MM-DD` (doctor own, admin all; patients 403): confirmed appointments of one clinic-local day in time order, previous/today/next links and a date picker, invalid date -> redirect with message; days cut at clinic midnight (DST-safe). Menu entry "Agenda du jour" (doctor, admin)
- Appointments step 4b (retention): deleting a doctor/patient is refused while they have upcoming confirmed appointments (cancel first); deactivated (user.active=false, record + history kept, doctor's free slots removed) when only past/cancelled appointments exist; hard-deleted only when no appointment ever referenced them. Deactivated doctors are hidden from search, cannot be booked or given slots; admin lists show a "Désactivé" badge and a Réactiver button. Lists skip appointments whose doctor/patient record is gone (legacy orphans)
- REST API 7a (`/api/v1`, docs in `docs/API.md`): JSON-only writes (415 otherwise, empty body ok), errors `{error:{status,message,fields?}}`, validation via `middleware/validate.js` (422 + per-field messages), explicit serializers (`utils/serializers.js`), shared `utils/session.js`. Endpoints: GET /health, POST /auth/login|logout|register, GET /auth/me; /specialties (CRUD), /doctors (CRUD, reactivate, `/:id/slots`; public view for non-admins, deactivated doctors hidden); /patients (admin CRUD + reactivate) and `/patients/me` (patient reads/edits own profile); /availability (doctor own, admin via `?doctor=`; list/create range/get/move/delete); /appointments (list, `/day`, get, book, modify, cancel; own appointments only, conflicts carry `fields.slot`). Same cookie session as the web app
- App shell: header + hamburger, collapsible role-based sidebar, footer, responsive (drawer < 62rem)
